// MARK: - Palette

enum P {
    // Graphite, warm.
    static let shell = Color.black
    static let base = hex(0x0D0D0C)
    static let panel = hex(0x131312)
    static let raised = hex(0x1A1A18)
    static let raised2 = hex(0x222220)
    static let rule = Color(red: 1, green: 0.98, blue: 0.94).opacity(0.075)
    static let ruleStrong = Color(red: 1, green: 0.98, blue: 0.94).opacity(0.13)
    // Ink.
    static let paper = hex(0xEDEBE6)
    static let pencil = hex(0xA3A09A)
    static let graphite = hex(0x6E6B66)
    static let faint = hex(0x4A4845)
    // Signals.
    static let beacon = hex(0xF5B544)   // needs you
    static let ember = hex(0xE0855A)    // working (Claude's clay)
    static let moon = hex(0xA9B6FF)     // done
    static let flare = hex(0xFF6B5E)    // stuck, failing
    static let moss = hex(0x7CC793)     // passing
    static func hex(_ v: UInt32) -> Color {
        Color(red: Double((v >> 16) & 0xFF) / 255, green: Double((v >> 8) & 0xFF) / 255, blue: Double(v & 0xFF) / 255)
    }
}

/// A lit lamp: the only colour on screen. Soft glow, so it reads on black like a signal light.
struct Lamp: View {
    let c: Color
    var size: CGFloat = 7
    var dim = false
    var body: some View {
        ZStack {
            if !dim { Circle().fill(c.opacity(0.28)).frame(width: size * 2.6, height: size * 2.6).blur(radius: size * 0.7) }
            Circle().fill(dim ? P.faint : c).frame(width: size, height: size)
            if !dim { Circle().fill(.white.opacity(0.35)).frame(width: size * 0.35, height: size * 0.35).offset(x: -size * 0.15, y: -size * 0.15) }
        }
        .frame(width: size * 2, height: size * 2)
    }
}

// MARK: - The island (squarer: 10 outside, 5 inside)

enum Notch { static let w: CGFloat = 188; static let h: CGFloat = 32 }

struct IslandShape: Shape {
    var r: CGFloat; var flare: CGFloat = 5
    func path(in b: CGRect) -> Path {
        let w = b.width, h = b.height, f = flare, R = min(r, h / 2)
        var p = Path()
        p.move(to: .zero)
        p.addQuadCurve(to: CGPoint(x: f, y: f), control: CGPoint(x: f, y: 0))
        p.addLine(to: CGPoint(x: f, y: h - R))
        p.addArc(tangent1End: CGPoint(x: f, y: h), tangent2End: CGPoint(x: f + R, y: h), radius: R)
        p.addLine(to: CGPoint(x: w - f - R, y: h))
        p.addArc(tangent1End: CGPoint(x: w - f, y: h), tangent2End: CGPoint(x: w - f, y: h - R), radius: R)
        p.addLine(to: CGPoint(x: w - f, y: f))
        p.addQuadCurve(to: CGPoint(x: w, y: 0), control: CGPoint(x: w - f, y: 0))
        p.closeSubpath(); return p
    }
}

// MARK: - Render

@main
struct Render {
    @MainActor static func main() {
        let dir = CommandLine.arguments.dropFirst().first ?? "shots"
        try? FileManager.default.createDirectory(atPath: dir, withIntermediateDirectories: true)
        func mid(_ w: CGFloat, _ h: CGFloat) -> CGRect { CGRect(x: (1512 - w) / 2, y: 0, width: w, height: h) }
        save(Sheet(), "0-identity", dir, nil)
        save(Rest(), "1-island-rest", dir, mid(720, 110))
        save(Arrives(), "2-island-question", dir, mid(720, 220))
        save(Glance(), "3-island-hover", dir, mid(720, 320))
        save(Hub(), "4-island-kept-open", dir, mid(760, 520))
        save(WindowShot(focus: .question), "5-router-question", dir, mid(1240, 880))
        save(WindowShot(focus: .plan), "6-router-plan", dir, mid(1240, 880))
        save(WindowShot(focus: .done), "7-router-done", dir, mid(1240, 880))
        save(WindowShot(focus: .chat), "8-router-chat", dir, mid(1240, 880))
    }

    @MainActor static func save<V: View>(_ v: V, _ name: String, _ dir: String, _ crop: CGRect?) {
        let r = ImageRenderer(content: v.environment(\.colorScheme, .dark))
        r.scale = 2
        guard var cg = r.cgImage else { print("failed \(name)"); return }
        if let crop, let c = cg.cropping(to: CGRect(x: crop.minX * 2, y: crop.minY * 2, width: crop.width * 2, height: crop.height * 2)) { cg = c }
        try? NSBitmapImageRep(cgImage: cg).representation(using: .png, properties: [:])?.write(to: URL(fileURLWithPath: "\(dir)/\(name).png"))
        print("\(dir)/\(name).png")
    }
}
